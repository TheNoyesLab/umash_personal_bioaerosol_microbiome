This is the code for paper: "The personal bioaerosol microbiome in swine farms: Particle-size-resolved characterization and comparison of two wearable sampling devices"

Goal 1: Describe the microbiome in 3 particle size ranges (stages) collected by NIOSH
Goal 2: Compare what Aerocollect capture with each NIOSH stage

This code analyze the species-level count matrix generated by emu. This data was generated by basecalling the reads with dorado and sup model, and filtered based on quality (Q12) and length (1.2-1.8kbp).

ps <- subset_samples(ps_swine, sample_or_control == "sample")
df <- data.frame(sample_data(ps))
emu_classif <- prune_taxa(taxa_names(ps) != "unassigned", ps)
emu_classif <- data.frame(sample_sums(emu_classif))
df <- cbind(df, emu_classif)
df <- df %>% dplyr::rename(emu_classif = sample_sums.emu_classif.)
# List of categorical variables
cat_vars <- c("flowcell","sample_type","Farm","DNAext_run","DNAextractor")
# Initialize empty table
table_list <- list()
for(var in cat_vars){
  # Compute median (Q1–Q3) for each category
  summary_stats <- df %>%
    group_by(across(all_of(var))) %>%
    summarise(
      n = n(),
      median_DNA = median(postPCR_DNA_conc, na.rm = TRUE),
      q1_DNA = quantile(postPCR_DNA_conc, 0.25, na.rm = TRUE),
      q3_DNA = quantile(postPCR_DNA_conc, 0.75, na.rm = TRUE),      
      median_reads = median(filt_Reads_total, na.rm = TRUE),
      q1_reads = quantile(filt_Reads_total, 0.25, na.rm = TRUE),
      q3_reads = quantile(filt_Reads_total, 0.75, na.rm = TRUE),
      median_classif = median(emu_classif, na.rm = TRUE),
      q1_classif = quantile(emu_classif, 0.25, na.rm = TRUE),
      q3_classif = quantile(emu_classif, 0.75, na.rm = TRUE),
      .groups = "drop"
    ) %>%
    mutate(
      median_iqr_DNA = paste0(round(median_DNA, 1)," [",
        round(q1_DNA, 1), " – ", round(q3_DNA, 1), "]"),
      median_iqr = paste0(round(median_reads, 1)," [",
        round(q1_reads, 1), " – ", round(q3_reads, 1), "]"),
      median_iqr_classif = paste0(round(median_classif, 1)," [",
        round(q1_classif, 1), " – ", round(q3_classif, 1),"]" ),
      category = paste0("   ", .data[[var]])
    ) %>%
    dplyr::select(category, n,median_iqr_DNA, median_iqr,median_iqr_classif)
  # Compute p-values
  fmla_DNA <- reformulate(var, response = "postPCR_DNA_conc")
  pval_DNA <- df %>%
    kruskal_test(formula = fmla_DNA) %>%
    pull(p) %>%
    signif(3)
  fmla <- reformulate(var, response = "filt_Reads_total")
  pval <- df %>%
    kruskal_test(formula = fmla) %>%
    pull(p) %>%
    signif(3)
  fmla_classif <- reformulate(var, response = "emu_classif")
  pval_classif <- df %>%
    kruskal_test(formula = fmla_classif) %>%
    pull(p) %>%
    signif(3)
  # Subheading row
  subheading <- tibble(
    category = var,
    n = NA,
    median_iqr_DNA = "",
    median_iqr = "",
    median_iqr_classif = "",
    p_DNA = as.character(pval_DNA),
    p_reads = as.character(pval),
    p_classif = as.character(pval_classif))
  # Add empty p-value columns to category rows
  summary_stats$p_DNA <- ""
  summary_stats$p_reads <- ""
  summary_stats$p_classif <- ""
  # Combine subheading + categories
  table_list[[var]] <- bind_rows(subheading, summary_stats)
}
