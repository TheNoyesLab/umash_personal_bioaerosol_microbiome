/scratch.global/diazo005/bc_models/dna_r10.4.1_e8.2_400bps_sup@v5.2.0 \
/scratch.global/diazo005/NN14/NN14_2026-01-22_FBA61501 \
--verbose \
--kit-name SQK-16S114-24 \
--emit-fastq \
--recursive \
--output-dir dorado_FBA61501

# RUNNING chopper (in a while loop)
# I will run chopper in mass. So I will go to the flow cell directory and then run a script to automatically 
# find fastq files, run chopper on them, modify their names adding "filtered" and output them into a "filtered" folder
# It will also print a log file with the number of reads that were filter from the original number of reads for every fastq file

cd /scratch.global/diazo005/dorado_NN14_2026_01_22

# Run
find . -type f -name "*.fastq" | while read -r fq; do
    dorado=$(echo "$fq" | awk -F/ '{for(i=1;i<=NF;i++) if($i ~ /^dorado_FBA/) {print $i; exit}}')
    outdir="filtered_${dorado}"
    mkdir -p "$outdir"

    base=$(basename "$fq" .fastq)
    echo "Processing: $base"

    chopper \
        --threads 40 \
        --quality 12 \
        --minlength 1200 \
        --maxlength 1800 \
        --input "$fq" \
        > "${outdir}/${base}_filtered.fastq"

done 2>&1 | tee chopper_filtering.log
